📦 Installation Guide

# Clone and build
git clone https://github.com/wynandw87/claude-code-youtube-mcp
cd claude-code-youtube-mcp
npm install && npm run build

# claude_desktop_config.json
{
  "mcpServers": {
    "youtube": {
      "command": "node",
      "args": ["C:/path/to/dist/index.js"],
      "env": { "YOUTUBE_API_KEY": "YOUR_KEY" }
    }
  }
}
# Install via uv tool
uv tool install mcp-youtube

# claude_desktop_config.json
{
  "mcpServers": {
    "youtube": {
      "command": "uv",
      "args": ["tool", "run", "mcp-youtube"]
    }
  }
}

# Config file (Windows):
# %APPDATA%\Claude\claude_desktop_config.json
# Install via Smithery (auto-configures Claude Desktop)
npx -y @smithery/cli install @ZubeidHendricks/youtube-mcp --client claude

# Restart Claude Desktop after installation
